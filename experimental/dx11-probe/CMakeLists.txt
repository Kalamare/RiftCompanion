cmake_minimum_required(VERSION 3.24)
project(RiftDx11Probe LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows and the Windows SDK are required")
endif()
add_executable(RiftDx11Probe WIN32 main.cpp)
set_property(TARGET RiftDx11Probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_features(RiftDx11Probe PRIVATE cxx_std_17)
target_compile_definitions(RiftDx11Probe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(RiftDx11Probe PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(RiftDx11Probe PRIVATE d3d11 dxgi d2d1 dwrite shell32)
if(GOVERLAY_SOURCE)
  set(GO "${GOVERLAY_SOURCE}/electron-overlay/src")
  target_sources(RiftDx11Probe PRIVATE bridge.cpp "${GO}/ipc/ipccenter.cc" "${GO}/ipc/ipclink.cc")
  target_include_directories(RiftDx11Probe PRIVATE "${GO}")
  target_compile_definitions(RiftDx11Probe PRIVATE WITH_GOVERLAY)
  # Upstream IPC sources rely on the original addon's precompiled includes.
  target_precompile_headers(RiftDx11Probe PRIVATE <Windows.h> <cstdint> <string> <map> <vector> <optional> <memory> <algorithm> <cassert>)
endif()
enable_testing()
add_test(NAME dx11-warp-lifecycle COMMAND RiftDx11Probe --smoke)
set_tests_properties(dx11-warp-lifecycle PROPERTIES TIMEOUT 30)
if(GOVERLAY_SOURCE)
  add_test(NAME goverlay-self-load-ipc COMMAND RiftDx11Probe --bridge-check)
  set_tests_properties(goverlay-self-load-ipc PROPERTIES TIMEOUT 30)
endif()

add_executable(RiftProbeDiagnostic diagnostic.cpp)
target_compile_features(RiftProbeDiagnostic PRIVATE cxx_std_17)
target_link_libraries(RiftProbeDiagnostic PRIVATE dbghelp)
